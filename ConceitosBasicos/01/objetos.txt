1. O que é um objeto?

# Uma classe é o modelo.
# Um objeto é uma instância desse modelo.

class Pessoa
{
    public string Nome;
    public int Idade;
}

# Isso é apenas a classe.
# Ainda não temos nenhuma pessoa criada.
# Quando fazemos:

Pessoa pessoa = new Pessoa();

# agora temos um objeto.
# Podemos imaginar:

Classe Pessoa
       ↓
   ┌─────────┐
   │ Pessoa  │  ← modelo
   └─────────┘
       ↓
      new
       ↓
   ┌─────────┐
   │ objeto  │
   │ Nome    │
   │ Idade   │
   └─────────┘

2. Podemos criar vários objetos da mesma classe

# Essa é uma das características mais importantes.

Pessoa pessoa1 = new Pessoa();
Pessoa pessoa2 = new Pessoa();
Pessoa pessoa3 = new Pessoa();

# Temos três objetos diferentes.3. O new é muito importante

# Quando você escreve:

Pessoa pessoa = new Pessoa();

# temos duas coisas acontecendo:
Pessoa pessoa
     ↓
variável que vai guardar uma referência para o objeto

new Pessoa()
     ↓
criação do objeto

# O new cria uma nova instância.
# Podemos colocar valores diferentes:


4. Objetos podem mudar

# Imagine:

Pessoa pessoa = new Pessoa();

pessoa.Nome = "Carlos";
pessoa.Idade = 28;

# Depois:

pessoa.Idade = 29;

# O objeto agora contém:

Nome: Carlos
Idade: 29

# Você alterou o estado do objeto.
# Esse termo é importante:
# Estado de um objeto = os valores que ele possui naquele momento.
