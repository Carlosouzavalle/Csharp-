# Uma classe é como um modelo que define como determinado objeto será.

# O que são essas coisas dentro da classe?

# Veja:
public class Game
{
    public int Id { get; set; }
    public string Title { get; set; }
    public string Genre { get; set; }
    public double Price { get; set; }
}

# Essas são propriedades.

# Por exemplo:

public string Title { get; set; }

# significa que o objeto Game possui uma propriedade chamada Title.

# Então podemos fazer:

game.Title = "Minecraft";

# O set permite alterar:

game.Title = "Minecraft";

# O get permite pegar o valor:

Console.WriteLine(game.Title);

using System;

{

        {
            // public string Nome { get; set; } = string.Empty;

                Console.WriteLine($"Nome: {Nome}");
                Console.WriteLine($"Idade: {Idade}");

        }

    }
}



1. Primeiro método — Object Initializer

Crie uma Pessoa e depois atribua esses valores às propriedades.

É como se, de forma simplificada, fosse:

Pessoa pessoa = new Pessoa();
pessoa.Nome = "Carlos";
pessoa.Idade = 28;
pessoa.Endereco = "Rua ABC123";


Você pode criar o objeto sem fornecer esses valores.
